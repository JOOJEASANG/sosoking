  // BEGIN FLIX ADMIN RULES
  function flixAdmin() {
   return signedIn() && exists(/databases/$(database)/documents/flixAdmins/$(request.auth.uid));
  }
  match /flixAdmins/{uid} {
   allow get: if signedIn() && request.auth.uid == uid;
   allow list, create, update, delete: if false;
  }
  match /flixPublic/catalog {
   allow get: if true;
   allow list, delete: if false;
   function validCatalog() {
    return request.resource.data.keys().hasOnly(['json','revision','updatedAt'])
      && request.resource.data.keys().hasAll(['json','revision','updatedAt'])
      && request.resource.data.json is string
      && request.resource.data.json.size() <= 700000
      && request.resource.data.revision is int
      && request.resource.data.updatedAt == request.time;
   }
   allow create: if flixAdmin() && validCatalog() && request.resource.data.revision == 1;
   allow update: if flixAdmin() && validCatalog() && request.resource.data.revision == resource.data.revision + 1;
  }
  // END FLIX ADMIN RULES
